  try(PrintWriter out=new PrintWriter(args[0]+"/jacobian.csv")){
   out.println("eos,rule,state,moles,T_K,P_bar,entry,analytical,coarse,fine,mixed_relative_error,gibbs_identity_mol,relative_pressure_residual");
   for(boolean pr:new boolean[]{false,true})for(int rule:new int[]{2,4})for(String state:new String[]{"gas","oil","aqueous"})for(double amount:new double[]{1,1000}){
